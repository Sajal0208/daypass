import { Card, CardGrid } from '@astrojs/starlight/components';

```bash
npm install -g peephole-db
peephole init
peephole open prod --ttl 45m --reason "why are campaign 1234's calls failing"
```

<CardGrid>
	<Card title="Read-only by construction" icon="approve-check">
		A dedicated Postgres role with `SELECT` on visible columns only, a read-only session and a
		10 second statement timeout. Hidden columns are hidden at the database, not in the client.
	</Card>
	<Card title="Time-boxed" icon="seti:clock">
		Access exists only while a lease is open. It expires on its own; nothing to remember to
		revoke.
	</Card>
	<Card title="No password in the chat" icon="seti:lock">
		On AWS RDS the role logs in with a short-lived IAM token minted from your own AWS identity.
	</Card>
	<Card title="Every query logged" icon="document">
		One JSON line per call: who, why, the SQL, rows returned, duration, allowed or blocked.
	</Card>
</CardGrid>
